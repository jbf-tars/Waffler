sounddevice>=0.4.6
numpy>=1.24.0
# FLAC encoding of longer clips before upload (libsndfile ships in the

soundfile>=0.12.1

# ── Speech to text on this computer (src/local_stt.py) ────────────────
# The engine only: the model (about 0.6 GB) is downloaded once, from
# Settings, into the data folder. docs/OFFLINE-TRANSCRIPTION.md has the why.
# Apple Silicon: whisper.cpp with Metal. Wheels: cp311 to cp314, macOS arm64.
pywhispercpp>=1.5.1; platform_machine == "arm64"
# Intel Macs: faster-whisper (CTranslate2). Not on Python 3.14 there: its
# onnxruntime dependency has no macOS x86_64 wheel for 3.14, and without the
# marker the whole install would fail. Waffler then says it has no engine.
faster-whisper>=1.2.0; platform_machine != "arm64" and python_version < "3.14"

# ── Tidying and notes on this computer (src/local_llm.py), optional ───
# llama.cpp through llama-cpp-python. NOT installed by this file: PyPI has
# no binary wheel for it, so pip compiles it (CMake comes with the build;
# Xcode's compiler must be there), and a Waffler without it simply does not
# offer the second download. To put it in a Mac release build, in the build's
# Python, before PyInstaller runs:
#   MACOSX_DEPLOYMENT_TARGET=14.0 \
#   CMAKE_ARGS="-DGGML_METAL=on -DCMAKE_OSX_DEPLOYMENT_TARGET=14.0" \
#     pip install "llama-cpp-python==0.3.37" --no-binary llama-cpp-python
# 0.3.37 is the version measured with the model (Gemma 4 E4B). Without the
# deployment target its libraries are built for the build machine's macOS
# only, and scripts/check_macos_minos.py fails the build. The model (5.2 GB)
# is never bundled. docs/OFFLINE-TRANSCRIPTION.md has the rest.

# ── Meetings: a library imported inside functions (src/meetings) ──
# Import audio (src/meetings/audio_decode.py): what soundfile cannot read (m4a,
# an iPhone voice memo) goes through macOS's own /usr/bin/afconvert. No FFmpeg
# package: no GPL program in the app (decided 26 Nov 2026).
# Meeting notes written by Anthropic's models (src/meetings/llm.py), when the
# person chooses Anthropic. Without it that choice fails at the import.
anthropic>=2.0.0
# keyring is left out on purpose (src/meetings/keys.py): keys reach meetings

# it is the one that ships. `Waffler --self-check` reports it.
